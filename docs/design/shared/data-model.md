# Shared Data Model

Cross-cutting tables used by multiple components.
PostgreSQL schema: `shared`.

## Tables

```sql
CREATE SCHEMA IF NOT EXISTS shared;

CREATE TABLE IF NOT EXISTS shared.t_api_usage (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    provider TEXT NOT NULL,    -- "finnhub" | "openai" | "marketaux"
    endpoint TEXT,
    tokens_used BIGINT,
    cost_estimate DOUBLE PRECISION,
    called_at TIMESTAMPTZ NOT NULL
);

CREATE TABLE IF NOT EXISTS shared.t_watchlist_tickers (
    ticker TEXT NOT NULL,
    exchange_code TEXT NOT NULL,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    source TEXT NOT NULL DEFAULT 'manual',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY (ticker, exchange_code)
);
```
